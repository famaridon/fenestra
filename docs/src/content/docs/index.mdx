  <Card title="Vertical displays" icon="laptop">

  </Card>
  <Card title="Shared border" icon="random">

  </Card>
  <Card title="Snap Layouts" icon="star">

  </Card>
  <Card title="Snap Assist" icon="puzzle">

  </Card>
  <Card title="Multiple displays" icon="external">
